# Hidden Intake Event
MTR_SWE.1001 = {
	hidden = yes
	
	# Mod Content "pack"
	content_source = dlc_MTR_SWE
	
	trigger = {
		#Make sure event is not disabled
		NOT = { has_game_rule = MTR_SWE_eligibility_GR_deactivated }
		MTR_SWE_trigger_is_valid_ward_gender = yes
		OR = {
			# Either training specifically martial...
			has_focus = education_martial

			# ...or has enough traits to warrant it...
			MTR_SWE_trigger_temperament = yes

			# ...or meets the prowess requirements in cultures that make exceptions for it...
			MTR_SWE_trigger_culture_allows_high_prowess = yes

			# ...or has a parent with the trait...
			MTR_SWE_trigger_either_parent_shieldperson = yes

			# ...or their is a game-rule removing the requirements
			has_game_rule = MTR_SWE_eligibility_GR_relaxed

		}
		OR = { #Is being trained/taught to some degree
			any_relation = {
				type = guardian
			}
			exists = court_owner.court_position:court_tutor_court_position
		}
	}

	immediate = {
		save_scope_as = child
		
		#Fire Event From Guardian's POV
		IF = {
			limit = {
				any_relation = {
					type = guardian
				}
			}
			random_relation = {
				type = guardian
				trigger_event = MTR_SWE.1002
			}
		}
		#Fire Event From Tutor's POV
		ELSE = {
			#At this point, if there is no guardian we know it should be the tutor
			court_owner.court_position:court_tutor_court_position ?= {
				trigger_event = MTR_SWE.1002
			}
		}
	}
}


# Guardian POV
MTR_SWE.1002 = {
	type = character_event
	title = MTR_SWE.1002.t
	#desc  = MTR_SWE.1002.desc
	theme = martial

	# Mod Content "pack"
	content_source = dlc_MTR_SWE

	left_portrait = {
		character = scope:child
		animation = marshal_random_weapon
	}

	immediate = { save_scope_as = guardian }

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { #Both Ward and guardian have cultures with shieldmaidens
					culture = { has_cultural_parameter = has_access_to_shieldmaidens }
				}
				desc = MTR_SWE.1002.start.culture_both
			}
			triggered_desc = {
				trigger = { #Guardian has culture without shieldmaidens
					NOT = { culture = { has_cultural_parameter = has_access_to_shieldmaidens } }
				}
				desc = MTR_SWE.1002.start.culture_child_only
			}
		}

		# Justify Martial Education
		triggered_desc = {
			trigger = { scope:child = { has_focus = education_martial } }
			desc = MTR_SWE.1002.desc.education_martial
		}

		# Justify Personality
		triggered_desc = {
			trigger = { scope:child = {MTR_SWE_trigger_temperament = yes} }
			desc = {
				desc = MTR_SWE.1002.desc.trait_intro

				triggered_desc = {
					trigger = { scope:child = { has_trait = ambitious } }
					desc = MTR_SWE.1002.desc.trait_ambitious
				}
				triggered_desc = {
					trigger = { scope:child = { has_trait = arrogant } }
					desc = MTR_SWE.1002.desc.trait_arrogant
				}
				triggered_desc = {
					trigger = { scope:child = { has_trait = brave } }
					desc = MTR_SWE.1002.desc.trait_brave
				}
				triggered_desc = {
					trigger = { scope:child = { has_trait = eccentric } }
					desc = MTR_SWE.1002.desc.trait_eccentric
				}
				triggered_desc = {
					trigger = { scope:child = { MTR_SWE_trigger_sadistic_and_relevant = yes } }
					desc = MTR_SWE.1002.desc.trait_sadistic
				}
				triggered_desc = {
					trigger = {
						scope:child = {
							OR = {
								has_trait = strong
								has_trait = physique_good_3
								has_trait = physique_good_2
							}
						}
					}
					desc = MTR_SWE.1002.desc.trait_strong
				}
				triggered_desc = {
					trigger = { scope:child = { MTR_SWE_trigger_vengeful_and_relevant = yes } }
					desc = MTR_SWE.1002.desc.trait_vengeful
				}
				triggered_desc = {
					trigger = { scope:child = { has_trait = wrathful } }
					desc = MTR_SWE.1002.desc.trait_wrathful
				}
				triggered_desc = {
					trigger = { scope:child = { MTR_SWE_trigger_zealous_and_relevant = yes} }
					desc = MTR_SWE.1002.desc.trait_zealous
				}

				first_valid = {
					triggered_desc = {
						trigger = { scope:child = { is_female = yes } }
						desc = MTR_SWE.1002.desc.trait_ending_female
					}
					triggered_desc = {
						trigger = { scope:child = { is_male = yes } }
						desc = MTR_SWE.1002.desc.trait_ending_male
					}
				}
			}
		}


		# Justify Parents' Legacy
		triggered_desc = {
			trigger = { scope:child = {MTR_SWE_trigger_either_parent_shieldperson = yes} }
				
			desc = {
				first_valid = {
					triggered_desc = {
						trigger = { scope:child = { has_trait = craven} }
						desc = MTR_SWE.1002.desc.parents_legacy_intro_scared
					}
					triggered_desc = {
						trigger = { scope:child = { NOT = { has_trait = craven } } }
						desc = MTR_SWE.1002.desc.parents_legacy_intro_normal
					}
				}
				desc = MTR_SWE.1002.desc.parents_legacy_ending
			}
		}
	}


	#Allow them to squire
	option = {
		name = MTR_SWE.1002.a

		#Duel if child is AI
		if = {
			limit = { scope:child = { is_ai = yes } }
			duel = {
				skills = { martial prowess }
				value = decent_skill_rating
				35 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = 2
					}
					opinion_modifier = {
						who = scope:child
						opinion_target = scope:guardian
						multiplier = 0.2
						step = 5
					}
					modifier = { #if guardian is a shieldperson, or a parent is, bump odds
						OR = {
							has_trait = shieldmaiden
							scope:child = { MTR_SWE_trigger_either_parent_shieldperson = yes}
						}
						add = 15
					}
					modifier = {
						scope:child = { has_trait = brave }
						add = 20
					}
					modifier = {
						scope:child = { has_trait = ambitious }
						add = 10
					}
					modifier = {
						scope:child = { has_trait = eccentric }
						add = 10
					}
					modifier = { #Sadistic tendencies reinforced by sadistic faiths
						scope:child = {
							MTR_SWE_trigger_sadistic_and_relevant = yes
						}
						add = 15
					}
					desc = MTR_SWE_child_squiring_success_effect.desc
					MTR_SWE_child_squiring_success_effect = yes
				}
				65 = {
					compare_modifier = {
						value = scope:duel_value
						multiplier = -2
					}
					opinion_modifier = {
						who = scope:child
						opinion_target = scope:guardian
						multiplier = -0.2
						step = 5
					}

					modifier = {
						has_game_rule = MTR_SWE_odds_GR_increased
						add = -30
					}

					modifier = { #Sadistic tendencies might be off-putting for non-sadistic faiths
						scope:child = {
							has_trait = sadistic
							faith = {
								NOT = {
									OR = {
										trait_is_virtue = sadistic
										has_doctrine = tenet_human_sacrifice
										has_doctrine = tenet_cranial_trophies
										has_doctrine = tenet_exaltation_of_pain
									}
								}
							}
						}
						add = 10
					}
					modifier = { #Arrogant people will have a tough time squiring
						scope:child = { has_trait = arrogant }
						add = 5
					}
					modifier = { #Shy afraid of prolonged time with someone as a squire
						scope:child = { has_trait = shy }
						add = 10
					}
					modifier = { #Cowards don't want to fight
						scope:child = { has_trait = craven }
						add = 20
					}
					desc = MTR_SWE_child_squiring_failure_effect.desc
					MTR_SWE_child_squiring_failure_effect = yes
				}
			}
		}
		else = { #if child is not AI, child has a choice (Trigger event from their POV)
			show_as_tooltip = {
				random_list = {
					1 = {
						show_chance = no
						desc = MTR_SWE_child_squiring_success_effect.desc
						MTR_SWE_child_squiring_success_effect = yes
					}
					1 = {
						show_chance = no
						desc = MTR_SWE_child_squiring_failure_effect.desc
						MTR_SWE_child_squiring_failure_effect = yes
					}
				}
			}

			scope:child = {
				trigger_event = MTR_SWE.1003
			}
		}
		
		ai_chance = {
			# Chance without the guardian being of a culture that allows shieldmaidens (1-to-1 odds)
			base = 25

			# If the guardian also belongs to a culture with shieldmaidens, make the chance higher (3-to-1 odds, instead of 1-to-1)
			modifier = {
				add = 50
				culture = { has_cultural_parameter = has_access_to_shieldmaidens }
			}

			# Just and Compassionate willing to entertain ward's unorthodox ambitions
			modifier = {
				add = 25
				has_trait = compassionate
			}
			modifier = {
				add = 25
				has_trait = just
			}
			# Eccentric likes breaking Status Quo
			modifier = {
				add = 25
				has_trait = eccentric
			}
		}
	}

	# Don't allow them to squire
	# Shieldmaidens will NEVER stop their wards from squiring if they are interested
	option = {
		name = MTR_SWE.1002.b

		ai_chance = {
			base = 25

			modifier = {
				add = -1000
				has_trait = shieldmaiden
			}

			modifier = { #Make more likely for player-controlled children
				add = -15
				scope:child = { is_ai = no }
			}
		}
	}
}
