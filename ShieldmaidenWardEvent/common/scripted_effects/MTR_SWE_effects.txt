# Handles toast message for when a ward tries to squire as a shieldmaiden and succeeds
MTR_SWE_child_squiring_success_effect = {
	scope:child = {
		# Effects for child
		add_trait = shieldmaiden
		stress_impact = {
			brave     = medium_stress_impact_loss
			arrogant  = minor_stress_impact_loss
			wrathful  = miniscule_stress_impact_loss
			eccentric = miniscule_stress_impact_loss
			ambitious = miniscule_stress_impact_loss
			craven    = major_stress_impact_gain
		}
		# Create the memory for the child
		create_character_memory = {
			type = MTR_SWE_memory
			participants = {
				child    = scope:child
				guardian = scope:guardian
			}
		}
		# Only people that care are the house-head and the parents
		house = {
			house_head = {
				add_to_list = MTR_SWE_success_message
			}
		}
		every_parent = {
			add_to_list = MTR_SWE_success_message
		}
	}
	every_in_list = {
		list = MTR_SWE_success_message
		limit = {
			this != scope:child
		}
		add_opinion = {
			target = scope:child
			modifier = impressed_opinion
			opinion = 20
		}
		send_interface_toast = {
			type = event_toast_effect_good
			title = MTR_SWE.1001.message_good
			left_icon = scope:child

			scope:child = {
				if = {
					limit = { has_trait = shieldmaiden }
					show_as_tooltip = { add_trait_force_tooltip = shieldmaiden }
				}
			}
		}
	}
}

# Handles toast message for when a ward tries to squire as a shieldmaiden and FAILS

	scope:child = {

			craven = minor_stress_impact_loss
		}
		# Create the memory for the child
		create_character_memory = {
			type = MTR_SWE_memory_fail
			participants = {
				child    = scope:child
				guardian = scope:guardian
			}
		}
		house = {
			house_head = {
				add_to_list = MTR_SWE_failure_message
			}
		}
		every_parent = {
			add_to_list = MTR_SWE_failure_message
		}
	}
	every_in_list = {
		list = MTR_SWE_failure_message
		limit = {
			this != scope:child
		}
		send_interface_toast = {
			type = event_toast_effect_bad
			title = MTR_SWE.1001.message_bad
			left_icon = scope:child
		}
	}
}
