# Trigger condensing the gendercheck for Shieldmaidens
MTR_SWE_trigger_is_valid_ward_gender = {
	#Ensure doesn't fire for adults
	is_adult = no

	#Gender Logic
	has_fp1_dlc_trigger = yes
	culture = {
		OR = {
			has_cultural_parameter = has_access_to_shieldmaidens
			AND = { #Gamerule to allow Fragile Peace / Bushido effect to use this event pack
				has_game_rule = MTR_SWE_widertraditions_GR_activated
				has_cultural_parameter = non_admin_close_family_non_martial_gender_knights_in_defensive_wars
			}
		}
	}
	trigger_if = { # Determined by Cultural Pillars
		limit = { has_dlc_feature = diverge_culture }
		OR = {
			AND = {
				is_female = yes
				culture = { has_cultural_parameter = martial_custom_male_only_combatant }
			}
			AND = {
				is_male = yes
				culture = { has_cultural_parameter = martial_custom_female_only_combatant }
			}
		}
	}
	trigger_else = {
		OR = { # Does not have the Royal Court and thus combatant is governed by faith
			AND = {
				is_female = yes
				faith = { has_doctrine_parameter = combatant_must_be_male_if_no_roco }
			}
			AND = {
				is_male = yes
				faith = { has_doctrine_parameter = combatant_must_be_female_if_no_roco }
			}
		}
	}
}

# Enough traits to warrant exception
MTR_SWE_trigger_temperament = {
	calc_true_if = {
		amount >= 3
		has_trait = ambitious
		has_trait = arrogant
		has_trait = brave
		has_trait = eccentric
		MTR_SWE_trigger_sadistic_and_relevant = yes
		MTR_SWE_trigger_vengeful_and_relevant = yes
		has_trait = wrathful
		MTR_SWE_trigger_zealous_and_relevant = yes
		has_trait = strong
		has_trait = physique_good_3
		has_trait = physique_good_2
	}
}

MTR_SWE_trigger_zealous_and_relevant = {
	AND = {
		has_trait = zealous
		faith = {
			OR = {
				has_doctrine = tenet_warmonger
				has_doctrine = tenet_unrelenting_faith
			}
		}
	}
}

MTR_SWE_trigger_vengeful_and_relevant = {
	AND = {
		has_trait = vengeful
		faith = {
			OR = {
				trait_is_virtue = vengeful
				has_doctrine = tenet_fp3_fedayeen
			}
		}
	}
}

MTR_SWE_trigger_sadistic_and_relevant = {
	AND = {
		has_trait = sadistic
		faith = {
			OR = {
				trait_is_virtue = sadistic
				has_doctrine = tenet_human_sacrifice
				has_doctrine = tenet_cranial_trophies
				has_doctrine = tenet_exaltation_of_pain
			}
		}
	}
}

#The child comes from a culture that enables high prowess to ignore restrictions and the child is strong enough
MTR_SWE_trigger_culture_allows_high_prowess = {
	OR = {
		AND = {
			prowess >= 10
			culture = { has_cultural_parameter = high_prowess_ignores_knight_restrictions }
			NOT = { culture = { has_cultural_parameter = minimum_prowess_for_knights } }
		}
		AND = {
			prowess >= 12
			culture = {
				has_cultural_parameter = high_prowess_ignores_knight_restrictions
				has_cultural_parameter = minimum_prowess_for_knights
			}
		}
	}
}

# Either Parent is a shieldperson
MTR_SWE_trigger_either_parent_shieldperson = {
	OR = {
		father ?= {has_trait = shieldmaiden}
		mother ?= {has_trait = shieldmaiden}
	}
}
